 boxrule=\RuleWeight,leftrule=3pt,arc=\ComponentRadius,before skip=10pt,after skip=10pt,
 attach boxed title to top left={yshift=-2.4mm,xshift=3mm},
 boxed title style={colback=#2!80!black,colframe=#2!80!black,
 boxrule=0pt,arc=\ComponentRadius,top=1mm,bottom=1mm,left=2.5mm,right=2.5mm},
 title={#3},fonttitle=\LabelFont\small}}
